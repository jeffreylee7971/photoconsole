from sqlalchemy import create_engine
from sqlalchemy.orm import Session
from photoconsole.catalog.models import Base, MediaFile
from photoconsole.dedup import find_duplicate_groups

engine = create_engine('sqlite://', future=True)
Base.metadata.create_all(engine)
with Session(engine) as s:
    for i, (sn, st) in enumerate([('D: SSD','local'),('OneDrive','rclone'),('iCloud','rclone')]):
        f = MediaFile(); f.path=f'p{i}'; f.hash='aaa'; f.source_name=sn; f.source_type=st; f.status='ok'
